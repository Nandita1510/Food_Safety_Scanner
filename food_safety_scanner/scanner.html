function analyzeText(rawText) { const text = rawText.toLowerCase(); let warnings = []; // 1. Sugar & Sweeteners if (text.includes("sugar") || text.includes("syrup") || text.includes("fructose") || text.includes("maltodextrin")) { warnings.push("🍬 High Sugar/Sweeteners: Detected sugar or high-glycemic syrups."); } // 2. Trans Fats if (text.includes("trans fat") || text.includes("hydrogenated") || text.includes("shortening")) { warnings.push("🚫 Trans Fats: Contains hydrogenated oils (unhealthy for heart)."); } // 3. Artificial Colors const colorList = ["red 40", "yellow 5", "yellow 6", "blue 1", "blue 2", "allura", "tartrazine", "sunset yellow"]; let foundColors = colorList.filter(color => text.includes(color)); if (foundColors.length > 0) { warnings.push(`🎨 Artificial Colors: Found ${foundColors.join(", ")}.`); } // 4. Preservatives/Sodium if (text.includes("sodium") || text.includes("benzoate") || text.includes("nitrite")) { warnings.push("🧂 Preservatives/Sodium: High salt or chemical preservatives detected."); } // Output formatting let output = `Extracted Text Preview:
${rawText.substring(0, 100)}...
`; if (warnings.length > 0) { output += warnings.map(w => `${w}`).join(""); } else { output += `

✔ Basic scan looks clean! No common red-flag ingredients found.
